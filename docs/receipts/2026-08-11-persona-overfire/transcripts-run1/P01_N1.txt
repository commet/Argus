```javascript
const router = express.Router();

// Stripe 웹훅은 raw body가 필요합니다 (서명 검증용)

const STRIPE_WEBHOOK_SECRET = process.env.STRIPE_WEBHOOK_SECRET;

router.post('/webhooks/stripe', (req, res) => {
  const sig = req.headers['stripe-signature'];

  let event;
  try {
    // TODO: stripe.webhooks.constructEvent(req.body, sig, STRIPE_WEBHOOK_SECRET) 사용
    event = req.body; // 임시 스텁: 실제 검증 로직으로 교체 필요
  } catch (err) {
    console.error('Webhook signature verification failed:', err.message);
    return res.status(400).send(`Webhook Error: ${err.message}`);
  }

  switch (event.type) {
    case 'payment_intent.succeeded':
      // TODO: 결제 성공 처리
      break;
    case 'payment_intent.payment_failed':
      // TODO: 결제 실패 처리
      break;
    case 'checkout.session.completed':
      // TODO: 체크아웃 완료 처리
      break;
    default:
      console.log(`Unhandled event type: ${event.type}`);
  }

  res.status(200).json({ received: true });
});

module.exports = router;
```

`app.js`에서 마운트할 때 예시:

```javascript

app.use('/api', express.raw({ type: 'application/json' }), webhookRouter);
```
