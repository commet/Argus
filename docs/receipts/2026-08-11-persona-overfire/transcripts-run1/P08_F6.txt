export function slugify(s) {
  return s.toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
}
---REPLY---
```javascript
import { describe, it, expect } from 'vitest';
import { slugify } from './slugify';

describe('slugify', () => {
  it('converts to lowercase', () => {
    expect(slugify('Hello World')).toBe('hello-world');
  });

  it('trims leading and trailing whitespace', () => {
    expect(slugify('  hello world  ')).toBe('hello-world');
  });

  it('replaces non-alphanumeric sequences with a single hyphen', () => {
    expect(slugify('hello___world!!')).toBe('hello-world');
    expect(slugify('foo@@@bar')).toBe('foo-bar');
  });

  it('collapses multiple consecutive separators into one hyphen', () => {
    expect(slugify('a   b---c___d')).toBe('a-b-c-d');
  });

  it('removes leading and trailing hyphens', () => {
    expect(slugify('---hello world---')).toBe('hello-world');
  });

  it('handles numbers correctly', () => {
    expect(slugify('Product 123 Name')).toBe('product-123-name');
  });

  it('returns an empty string when input has no alphanumeric characters', () => {
    expect(slugify('!!!')).toBe('');
  });

  it('returns an empty string for an empty input', () => {
    expect(slugify('')).toBe('');
  });

  it('handles already-slugified strings unchanged', () => {
    expect(slugify('already-slugified')).toBe('already-slugified');
  });

  it('handles unicode letters by stripping them (non a-z0-9 chars)', () => {
    expect(slugify('héllo wörld')).toBe('h-llo-w-rld');
  });
});
```
