export function slugify(s) {
  return s.toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
}
---REPLY---
```js
import { describe, it, expect } from 'vitest';
import { slugify } from './slugify';

describe('slugify', () => {
  it('converts to lowercase', () => {
    expect(slugify('Hello World')).toBe('hello-world');
  });

  it('trims leading and trailing whitespace', () => {
    expect(slugify('  Hello World  ')).toBe('hello-world');
  });

  it('replaces non-alphanumeric sequences with a single hyphen', () => {
    expect(slugify('foo   bar!!baz')).toBe('foo-bar-baz');
  });

  it('removes leading and trailing hyphens', () => {
    expect(slugify('--hello world--')).toBe('hello-world');
  });

  it('handles strings with numbers', () => {
    expect(slugify('Item 42 - Special Edition')).toBe('item-42-special-edition');
  });

  it('handles underscores and symbols', () => {
    expect(slugify('foo_bar@baz.com')).toBe('foo-bar-baz-com');
  });

  it('returns an empty string for an all-symbol input', () => {
    expect(slugify('!!!')).toBe('');
  });

  it('returns an empty string for empty input', () => {
    expect(slugify('')).toBe('');
  });

  it('handles already-slugified strings unchanged', () => {
    expect(slugify('already-a-slug')).toBe('already-a-slug');
  });

  it('handles unicode letters by stripping them (non a-z0-9)', () => {
    expect(slugify('héllo wörld')).toBe('h-llo-w-rld');
  });
});
```
